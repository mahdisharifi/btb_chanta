-- ============================================
-- 1.1 USERS
-- ============================================
CREATE TABLE users (
    id BIGSERIAL PRIMARY KEY,
    uuid UUID UNIQUE NOT NULL DEFAULT gen_random_uuid(),
    user_type VARCHAR(20) NOT NULL CHECK (user_type IN ('supplier','retailer_special','retailer','customer','admin')),
    email VARCHAR(255) UNIQUE NOT NULL,
    mobile VARCHAR(15) UNIQUE,
    password VARCHAR(255) NOT NULL,
    first_name VARCHAR(100),
    last_name VARCHAR(100),
    national_code VARCHAR(10),
    avatar VARCHAR(500),
    is_active BOOLEAN DEFAULT TRUE,
    is_verified BOOLEAN DEFAULT FALSE,
    email_verified_at TIMESTAMP,
    mobile_verified_at TIMESTAMP,
    last_login_at TIMESTAMP,
    last_login_ip INET,
    created_at TIMESTAMP DEFAULT NOW(),
    updated_at TIMESTAMP DEFAULT NOW(),
    deleted_at TIMESTAMP
);

--============================================
-- 1.2 USER PROFILES (اطلاعات تکمیلی)
-- ============================================
CREATE TABLE user_profiles (
    id BIGSERIAL PRIMARY KEY,
    user_id BIGINT UNIQUE NOT NULL REFERENCES users(id) ON DELETE CASCADE,
    gender VARCHAR(10),
    birth_date DATE,
    bio TEXT,
    website VARCHAR(500),
    telegram(100),
    whatsapp VARCHAR(20),
    instagram VARCHAR(100),
    created_at TIMESTAMP DEFAULT NOW(),
    updated_at TIMESTAMP DEFAULT NOW()
);

-- ============================================
-- 1.3 USER ADDRESSES
-- ============================================
CREATE TABLE user_addresses (
    id BIGSERIAL PRIMARY KEY,
    user_id BIGINT NOT NULL REFERENCES users(id) ON DELETE CASCADE,
    title VARCHAR(100), -- دفتر مرکزی، انبار، ...
    province_id BIGINT NOT NULL,
    city_id BIGINT NOT NULL,
    postal_code VARCHAR(10),
    full_address TEXT NOT NULL,
    latitude DECIMAL(10,8),
    longitude DECIMAL(11,8),
    receiver_name VARCHAR(200),
    receiver_mobile VARCHAR(15),
    is_default BOOLEAN DEFAULT FALSE,
    created_at TIMESTAMP DEFAULT NOW(),
    updated_at TIMESTAMP DEFAULT NOW(),
    deleted_at TIMESTAMP
);

-- ============================================
-- 1.4 OTP CODES
-- ============================================
CREATE TABLE otp_codes (
    id BIGSERIAL PRIMARY KEY,
    user_id BIGINT REFERENCES users(id) ON DELETE CASCADE,
    mobile VARCHAR(15),
    email VARCHAR(255),
    code VARCHAR(6) NOT NULL,
    type VARCHAR(20) NOT NULL, -- login, register, reset_password, verify
    expires_at TIMESTAMP NOT NULL,
    used_at TIMESTAMP,
    attempts INT DEFAULT 0,
    created_at TIMESTAMP DEFAULT NOW()
);

-- ============================================
-- 1.5 USER SESSIONS
-- ============================================
CREATE TABLE user_sessions (
    id BIGSERIAL PRIMARY KEY,
    user_id BIGINT NOT NULL REFERENCES users(id) ON DELETE CASCADE,
    token_hash VARCHAR(255) NOT NULL,
    device_name VARCHAR(200),
    device_type VARCHAR(50), -- mobile, desktop, tablet
    os VARCHAR(100),
    browser VARCHAR(100),
    ip_address INET,
    last_activityat TIMESTAMP,
    expires_at TIMESTAMP,
    created_at TIMESTAMP DEFAULT NOW()
);

-- ============================================
-- 1.6 ROLES (spatie compatible)
-- ============================================
CREATE TABLE roles (
    id BIGSERIAL PRIMARY KEY,
    name VARCHAR(125) NOT NULL,
    guard_name VARCHAR(125) NOT NULL DEFAULT 'web',
    display_name VARCHAR(200),
    description TEXT,
    is_system BOOLEAN DEFAULT FALSE,
    created_at TIMESTAMP DEFAULT NOW(),
    updated_at TIMESTAMP DEFAULT NOW(),
    UNIQUE(name, guard_name)
);

-- ============================================
-- 1.7 PERMISSIONS (spatie compatible)
-- ================================
CREATE TABLE permissions (
    id BIGSERIAL PRIMARY KEY,
    name VARCHAR(125) NOT NULL,
    guard_name VARCHAR(125) NOT NULL DEFAULT 'web',
    group_name VARCHAR(100),
    display_name VARCHAR(200),
    created_at TIMESTAMP DEFAULT NOW(),
    updated_at TIMESTAMP DEFAULT NOW(),
    UNIQUE(name, guard_name)
);

-- ============================================
-- 1.8 MODEL_HAS_ROLES
-- ============================================
CREATE TABLE model_has_roles (
    role_id BIGINT NOT NULL REFERENCES roles(id) ON DELETE CASCADE,
    model_type VARCHAR(255) NOT NULL,
    model_id BIGINT NOT NULL,
    PRIMARY KEY (role_id, model_id, model_type)
);

-- ============================================
-- 1.9 MODEL_HAS_PERMISSIONS
-- ============================================
CREATE TABLE model_has_permissions (
    permission_id BIGINT NOT NULL REFERENCES permissions(id) ON DELETE CASCADE,
    model_type VARCHAR(255) NOT NULL,
    model_id BIGINT NOT NULL,
    PRIMARY KEY (permission_id, model_id, model_type)
);

-- ============================================
-- 1.10 ROLE_HAS_PERMISSIONS
-- ================================
CREATE TABLE role_has_permissions (
    permission_id BIGINT NOT NULL REFERENCES permissions(id) ON DELETE CASCADE,
    role_id BIGINT NOT NULL REFERENCES roles(id) ON DELETE CASCADE,
    PRIMARY KEY (permission_id, role_id)

-- ============================================

CREATE TABLE supplier_profiles (
    id BIGSERIAL PRIMARY KEY,
    user_id BIGINT UNIQUE NOT NULL REFERENCES users(id) ON DELETE CASCADE,
    company_name VARCHAR(255) NOT NULL,
    company_name_en VARCHAR(255),
    brand_name VARCHAR(200),

    established_year SMALLINT,
    employees_count VARCHAR(20), -- 1-10, 11-50, 51-200, 200+

    logo VARCHAR(500),
    banner VARCHAR(500),
    slug VARCHAR(255) UNIQUE NOT NULL,
    min_order_amount DECIMAL(15,0) DEFAULT 0,
    commission_rate DECIMAL(5,2) DEFAULT 0, -- درصد کمیسیون پلتفرم
    rating DECIMAL(3,2) DEFAULT 0,
    total_reviews INT DEFAULT 0,
    total_products INT DEFAULT 0,
    total_sales BIGINT DEFAULT 0,
    is_verified BOOLEAN DEFAULT FALSE,
    is_featured BOOLEAN DEFAULT FALSE,
    verified_at TIMESTAMP,
    status VARCHAR(20) DEFAULT 'pending', -- pending, active, suspended, rejected
    rejection_reason TEXT,
    created_at TIMESTAMP DEFAULT NOW(),
    updated_at TIMESTAMP DEFAULT NOW(),
    deleted_at TIMESTAMP

    reviewed_at TIMESTAMP,

    expires_at DATE,

-- 2.5 SUPPLIER BRANDS (برندهای مجاز)

-- ================================
-- 2.6 SUPPLIER SETTLEMENTS (تسویه حساب)

CREATE TABLE supplier_settlements (

    bank_account_id BIGINT NOT NULL REFERENCES supplier_bank_accounts(id),
    amount DECIMAL(15,0) NOT NULL,
    commission_amount DECIMAL(15,0) DEFAULT 0,
    net_amount DECIMAL(15,0) NOT NULL,
    period_from DATE NOT NULL,
    period_to DATE NOT NULL,
    status VARCHAR(20) DEFAULT 'pending', -- pending, processing, paid, failed
    transaction_ref VARCHAR(100),
    paid_at TIMESTAMP,
    description TEXT,
    created_at TIMESTAMP DEFAULT NOW(),
    updated_at TIMESTAMP DEFAULT NOW()
);

-- ============================================
-- 5.1 CATEGORIES (Nested Set / Adjacency List)
-- ============================================
CREATE TABLE categories (
    id BIGSERIAL PRIMARY KEY,
    parent_id BIGINT REFERENCES categories(id) ON DELETE SET NULL,
    name VARCHAR(200) NOT NULL,
    slug VARCHAR(200) UNIQUE NOT NULL,
    description TEXT,
    icon VARCHAR(100),
    image VARCHAR(500),
    meta_title VARCHAR(200),
    meta_description VARCHAR(500),
    depth SMALLINT DEFAULT 0,
    sort_order INT DEFAULT 0,
    is_active BOOLEAN DEFAULT TRUE,
    is_featured BOOLEAN DEFAULT FALSE,
    products_count INT DEFAULT 0,
    -- Nested Set columns
    _lft INT DEFAULT 0,
    _rgt INT DEFAULT 0,
    created_at TIMESTAMP DEFAULT NOW(),
    updated_at TIMESTAMP DEFAULT NOW(),
    deleted_at TIMESTAMP
);

-- ============================================
-- 5.2 BRANDS
-- ============================================
CREATE TABLE brands (
    id BIGSERIAL PRIMARY KEY,
    name VARCHAR(200) NOT NULL,
    name_en VARCHAR(200),
    slug VARCHAR(200) UNIQUE NOT NULL,
    logo VARCHAR(500),
    description TEXT,
    website VARCHAR(500),
    country(100),
    is_active BOOLEAN DEFAULT TRUE,
    is_featured BOOLEAN DEFAULT FALSE,
    sort_order INT DEFAULT 0,
    meta_title VARCHAR(200),
    meta_description VARCHAR(500),
    created_at TIMESTAMP DEFAULT NOW(),
    updated_at TIMESTAMP DEFAULT NOW(),
    deleted_at TIMESTAMP
);

-- ================================
-- 5.3 UNITS (واحد اندازه‌گیری)
-- ============================================
CREATE TABLE units (
    id BIGSERIAL PRIMARY KEY,
    name VARCHAR(100) NOT NULL, -- عد، کارتن، بسته، کیلوگرم
    symbol VARCHAR(20),
    type VARCHAR(50), -- count, weight, volume, length
    is_active BOOLEAN DEFAULT TRUE
);

-- ============================================
-- 5.4 PRODUCTS
-- ============================================
CREATE TABLE products (
    id BIGSERIAL PRIMARY KEY,
    uuid UUID UNIQUE NOT NULL DEFAULT gen_random_uuid(),
    supplier_id BIGINT NOT NULL REFERENCES supplier_profiles(id),
    category_id BIGINT NOT NULL REFERENCES categories(id),
    brand_id BIGINT REFERENCES brands(id),
    unit_id BIGINT REFERENCES units(id),
    title VARCHAR(500) NOT NULL,
    title_en VARCHAR(500),
    slug VARCHAR(500) UNIQUE NOT NULL,
    sku VARCHAR(100) UNIQUE,
    barcode VARCHAR(50),
    iran_code VARCHAR(20), -- کد ایران‌کد
    short_description VARCHAR(1000),
    description TEXT,
    -- Packaging Info
    items_per_carton INT, -- تعداد در کارتن
    carton_weight DECIMAL(10,3), -- وزن کارتن (kg)
    item_weight DECIMAL(10,3), -- وزن واحد (kg)
    -- Dimensions
    length DECIMAL(10,2), -- cm
    width DECIMAL(10,2),
    height DECIMAL(10,2),
    -- Status
    status VARCHAR(20) DEFAULT 'draft', -- draft, pending, active, inactive, rejected
    rejection_reason TEXT,
    is_active BOOLEAN DEFAULT TRUE,
    is_featured BOOLEAN DEFAULT FALSE,
    is_new BOOLEAN DEFAULT FALSE,
    -- Counters
    views_count INT DEFAULT 0,
    sales_count INT DEFAULT 0,
    rating DECIMAL(3,2) DEFAULT 0,
    reviews_count INT DEFAULT 0,

    -- SEO
    meta_title VARCHAR(200),
    meta_description VARCHAR(500),
    meta_keywords VARCHAR(500),

    -- Timestamps
    published_at TIMESTAMP,
    created_at TIMESTAMP DEFAULT NOW(),
    updated_at TIMESTAMP DEFAULT NOW(),
    deleted_at TIMESTAMP
);

CREATE INDEX idx_products_supplier ON products(supplier_id);
CREATE INDEX idx_products_category ON products(category_id);
CREATE INDEX idx_products_brand ON products(brand_id);
CREATE INDEX idx_products_status ON products(status);
CREATE INDEX idx_products_slug ON products(slug);
CREATE INDEX idx_products_sku ON products(sku);

sql
-- ============================================
-- 6.1 PRODUCT PRICES (قیمت‌های تامین‌کننده)
-- ============================================
CREATE TABLE product_prices (
    id BIGSERIAL PRIMARY KEY,
    product_id BIGINT NOT NULL REFERENCES products(id) ON DELETE CASCADE,
    variant_id BIGINT REFERENCES product_variants(id) ON DELETE CASCADE,
    supplier_id BIGINT NOT NULL REFERENCES supplier_profiles(id),
    -- قیمت‌های مختلف
    consumer_price DECIMAL(15,0), -- قیمت مصرف‌کننده
    base_price DECIMAL(15,0) NOT NULL, -- قیمت پایه (تکی)
    carton_price DECIMAL(15,0), -- قیمت کارتنی
    half_carton_price DECIMAL(15,0), -- قیمت نیم کارتنی
    -- قیمت‌های اعتباری
    credit_1_monthprice DECIMAL(15,0),
    credit_2_month_price DECIMAL(15,0),
    credit_3_month_price DECIMAL(15,0),
    -- تخفیف
    discount_percent DECIMAL(5,2) DEFAULT 0,
    discount_amount DECIMAL(15,0) DEFAULT 0,
    discounted_price DECIMAL(15,0),
    discount_starts_at TIMESTAMP,
    discount_ends_at TIMESTAMP,

    -- حداقل/حداکثر سفارش
    min_order_quantity INT DEFAULT 1,
    max_order_quantity INT,

    -- واحد قیمت
    currency VARCHAR(3) DEFAULT 'IR',

    -- اعتبار قیمت
    valid_from TIMESTAMP DEFAULT NOW(),
    valid_until TIMESTAMP,

    is_active BOOLEAN DEFAULT TRUE,
    created_at TIMESTAMP DEFAULT NOW(),
    updated_at TIMESTAMP DEFAULT NOW()
);

CREATE INDEX idx_prices_product ON product_prices(product_id);
CREATE INDEX idx_prices_supplier ON product_prices(supplier_id);
CREATE INDEX idx_prices_active ON product_prices(is_active, valid_from, valid_until);

نکات معماری مهم
Soft Delete روی جداول حیاتی (users, products, orders, suppliers) برای حفظ یکپارچگی داده.
Snapshot Pattern در سفارشات — آدرس و اطلاعات محصول در لحظه ثبتذخیره می‌شود.
Generated Columns مثل available_quantity در inventory برای کهش محاسبات runtime.
JSONB برای داده‌های نیمه‌ساختاریافته (metadata, gateway_response, attachments).
UUID برای شناسه‌های عمومی (API-facing) و BIGSERIAL برای کلیدهای داخلی.
Composite Indexes روی فیلدهای پرکاربرد در queryها.
Partitioning پیشنهادی روی order_payments, activity_logs, product_views بر اساس تاریخ.
