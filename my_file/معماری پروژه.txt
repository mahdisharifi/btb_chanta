-- ============================================
-- USERS & AUTHENTICATION
-- ============================================

CREATE TABLE users (
    id BIGSERIAL PRIMARY KEY,
    uuid UUID UNIQUE NOT NULL DEFAULT gen_random_uuid(),
    user_type VARCHAR(20) NOT NULL, -- supplier, retailer_special, retailer, customer
    email VARCHAR(255) UNIQUE NOT NULL,
    mobile VARCHAR(15) UNIQUE,
    password VARCHAR(255) NOT NULL,
    first_name VARCHAR(100),
    last_name VARCHAR(100),
    national_code VARCHAR(10) UNIQUE,
    is_verified BOOLEAN DEFAULT FALSE,
    is_active BOOLEAN DEFAULT TRUE,
    email_verified_at TIMESTAMP,
    mobile_verified_at TIMESTAMP,
    last_login_at TIMESTAMP,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    deleted_at TIMESTAMP
);

CREATE INDEX idx_users_type ON users(user_type);
CREATE INDEX idx_users_email ON users(email);
CREATE INDEX idx_users_mobile ON users(mobile);

معماری کامل پلتفرم B2B در سطح استارتاپ‌های بزرگ
1. معماری کلان سیستم (High-Level Architecture)
الگوی پیشنهادی: Modular Monolith قابل مهاجرت به Microservices

در مراحل اولیه استارتاپ بهتر است سیستم به صورت Modular Monolith در Laravel ساخته شود تا پیچیدگی DevOps کم بماند، ولی ساختار طوری طراحی شود که در آینده هر ماژول به Microservice تبدیل شود.

ساختار منطقی:

Client Layer

→ Web App (Livewire)

→ Mobile App (API)

API Gateway Layer

→ Laravel API

Application Layer

→ Domain Modules

Infrastructure Layer

→ Database

→ Cache

→ Queue

→ Search

→ Storage

نمای معماری:

User

→ CDN

→ Nginx Load Balancer

→ Laravel Application Nodes

→ Redis

→ PostgreSQL Cluster

→ Search Engine

→ Object Storage
